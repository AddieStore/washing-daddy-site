// Firebase Storage rules addition (Storage > Rules). Washers upload proof photos; customers/admin read via download URL.
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /washphotos/{wash}/{file} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.token.phone_number is string
                   && request.resource.size < 1500000
                   && request.resource.contentType.matches('image/.*');
    }
  }
}
